    #[cfg(target_os = "linux")]
    #[tokio::test]
    #[ignore = "isolated ARM64 VM only: requires privileged TCP+UDP TPROXY and local SOCKS5 relay; 480-second soak"]
    async fn privileged_arm64_mixed_tcp_udp_soak() {
        use std::sync::Arc;
        use std::time::Duration;
        use crate::model::LocalProxyProfile;
        use tokio_util::sync::CancellationToken;
        let config = Arc::new(LocalProxyProfile {
            id: "soak-lab".into(), name: "isolated-soak".into(), port: 47996,
            username: String::new(), password: String::new(),
        });
        let _ = std::fs::remove_file("/tmp/csqtt-soak-client.verdict");
        let sockets = super::bind_sockets(47997).expect("bind real TPROXY listeners");
        let cancel = CancellationToken::new();
        let stats = Arc::new(TproxyStats::default());
        let log: crate::proxy_route::LogFn = Arc::new(|level, message| println!("TPROXY {level}: {message}"));
        let task = tokio::spawn(super::run(sockets, config, cancel.clone(), log, stats.clone(), DeviceQuotaRegistry::new()));
        // Capture UDP association pressure every second to correlate short loss bursts.
        for second in 1..=480 {
            tokio::time::sleep(Duration::from_secs(1)).await;
            let sample = stats.snapshot();
            println!("SOAK_SAMPLE second={} tcp_active={} udp_active={} udp_peak={} udp_total={} udp_budget_rejects={} tcp_budget_rejects={}",
                second, sample.tcp_active, sample.udp_active, sample.udp_peak,
                sample.udp_total, sample.udp_budget_rejects, sample.tcp_budget_rejects);
        }
        let snapshot = stats.snapshot();
        println!("SOAK_NETWORK_TCP={} UDP={} UDP_PEAK={} UDP_ACTIVE={} UDP_BUDGET_REJECTS={} TCP_BUDGET_REJECTS={}",
            snapshot.tcp_total, snapshot.udp_total, snapshot.udp_peak,
            snapshot.udp_active, snapshot.udp_budget_rejects, snapshot.tcp_budget_rejects);
        cancel.cancel();
        let counts = tokio::time::timeout(Duration::from_secs(10), task).await.expect("TPROXY task stopped").unwrap();
        assert!(counts.0 >= 500 && counts.2 >= 500, "expected sustained real TCP and UDP flows");
        let verdict = std::fs::read_to_string("/tmp/csqtt-soak-client.verdict")
            .expect("client must write independent success/failure verdict");
        assert_eq!(verdict.trim(), "PASS", "real network client reported failed requests");
    }
